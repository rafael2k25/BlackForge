CREATE TABLE dbo.ApuracoesCustoOS (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    OrdemServicoId INT NOT NULL,
    Versao INT NOT NULL,
    DataApuracao DATETIME2(0) NOT NULL,
    CustoMaterialReal DECIMAL(18,2) NOT NULL,
    CustoMaquinaReal DECIMAL(18,2) NOT NULL,
    CustoOperadorReal DECIMAL(18,2) NOT NULL,
    CustoSetupReal DECIMAL(18,2) NOT NULL,
    CustoIndiretoReal DECIMAL(18,2) NOT NULL,
    CustoTotalReal DECIMAL(18,2) NOT NULL,
    ReceitaReal DECIMAL(18,2) NOT NULL,
    ResultadoReal AS CONVERT(DECIMAL(18,2), ReceitaReal - CustoTotalReal) PERSISTED,
    MargemRealPercentual AS CONVERT(DECIMAL(9,3),
        CASE WHEN ReceitaReal > 0
        THEN ((ReceitaReal - CustoTotalReal) * 100.0) / ReceitaReal
        END) PERSISTED,
    Status NVARCHAR(20) NOT NULL,
    Observacoes NVARCHAR(1000) NULL,
    CONSTRAINT UQ_ApuracoesCustoOS_Versao UNIQUE (OrdemServicoId, Versao)
);

CREATE TABLE dbo.CustosOperador (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    FuncionarioId INT NULL,
    Cargo NVARCHAR(100) NULL,

    HorasProdutivasMes DECIMAL(18,2) NOT NULL,
    CustoHora AS CONVERT(DECIMAL(18,4),

    Ativo BIT NOT NULL,

-- =============================================
-- RESTRIÇÕES CHECK
-- =============================================

ALTER TABLE dbo.ApuracoesCustoOS
ADD CONSTRAINT CK_ApuracoesCustoOS_Status
CHECK (Status IN ('REVISADA', 'FECHADA', 'RASCUNHO'));

ALTER TABLE dbo.ApuracoesCustoOS
ADD CONSTRAINT CK_ApuracoesCustoOS_Valores
CHECK (
    CustoMaterialReal >= 0 AND CustoMaquinaReal >= 0
    AND CustoOperadorReal >= 0 AND CustoSetupReal >= 0
    AND CustoIndiretoReal >= 0 AND CustoTotalReal >= 0
    AND ReceitaReal >= 0
);

ALTER TABLE dbo.ApuracoesCustoOS
ADD CONSTRAINT CK_ApuracoesCustoOS_Versao
CHECK (Versao > 0);

ALTER TABLE dbo.ConfiguracoesPrecificacao
ADD CONSTRAINT CK_ConfigPrec_Percentuais
CHECK (
    MargemDesejadaPercentual >= 0 AND MargemDesejadaPercentual < 100
    AND ImpostosPercentual >= 0 AND ImpostosPercentual < 100
    AND DespesasVariaveisPercentual >= 0 AND DespesasVariaveisPercentual < 100
    AND CustosIndiretosPercentual >= 0
);

ALTER TABLE dbo.ConfiguracoesPrecificacao
ADD CONSTRAINT CK_ConfigPrec_Valores
CHECK (SetupPadraoMinutos >= 0 AND CustoHoraOperadorPadrao >= 0);

ALTER TABLE dbo.CustosMaquina
ADD CONSTRAINT CK_CustosMaquina_Metodo
CHECK (MetodoCalculo IN ('MANUAL', 'DETALHADO'));

ALTER TABLE dbo.CustosMaquina
ADD CONSTRAINT CK_CustosMaquina_Valores
CHECK (
    ValorAquisicao >= 0 AND ValorResidual >= 0
    AND ValorResidual <= ValorAquisicao AND VidaUtilHoras >= 0
    AND ConsumoEnergiaKW >= 0 AND CustoKWh >= 0
    AND CustoManutencaoHora >= 0 AND OutrosCustosHora >= 0
    AND (CustoHoraManual IS NULL OR CustoHoraManual >= 0)
);

ALTER TABLE dbo.CustosOperador
ADD CONSTRAINT CK_CustosOperador_Identificacao
CHECK (
    (FuncionarioId IS NOT NULL AND Cargo IS NULL)
    OR (FuncionarioId IS NULL AND Cargo IS NOT NULL)
);

ALTER TABLE dbo.CustosOperador
ADD CONSTRAINT CK_CustosOperador_Valores
CHECK (CustoMensalTotal >= 0 AND HorasProdutivasMes > 0);

ALTER TABLE dbo.ItensPrecificacao
ADD CONSTRAINT CK_ItensPrecificacao_Tipo
CHECK (TipoItem IN ('OUTRO', 'INDIRETO', 'SETUP', 'OPERADOR', 'MAQUINA', 'MATERIAL'));

ALTER TABLE dbo.ItensPrecificacao
ADD CONSTRAINT CK_ItensPrecificacao_Valores
CHECK (Quantidade >= 0 AND CustoUnitario >= 0 AND CustoTotal >= 0);

ALTER TABLE dbo.Precificacoes
ADD CONSTRAINT CK_Precificacoes_Custos
CHECK (
    CustoMaterialEstimado >= 0 AND CustoMaquinaEstimado >= 0
    AND CustoOperadorEstimado >= 0 AND CustoSetupEstimado >= 0
    AND CustoIndiretoEstimado >= 0 AND CustoTotalEstimado >= 0
    AND PrecoSugerido >= 0
    AND (PrecoAprovado IS NULL OR PrecoAprovado >= 0)
);

ALTER TABLE dbo.Precificacoes
ADD CONSTRAINT CK_Precificacoes_Quantidade
CHECK (QuantidadeConsiderada > 0);

ALTER TABLE dbo.Precificacoes
ADD CONSTRAINT CK_Precificacoes_Status
CHECK (Status IN ('SUBSTITUIDA', 'REJEITADA', 'APROVADA', 'RASCUNHO'));
