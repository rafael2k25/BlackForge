/* ============================================================
   1. MATERIAIS
   ============================================================ */

CREATE TABLE Materiais
(
    Id INT IDENTITY(1,1) NOT NULL,

    Codigo NVARCHAR(50) NOT NULL,
    Nome NVARCHAR(150) NOT NULL,
    Categoria NVARCHAR(50) NOT NULL,
    Unidade NVARCHAR(20) NOT NULL,
    Descricao NVARCHAR(500) NULL,

    EstoqueMinimo DECIMAL(18,2) NOT NULL
        CONSTRAINT DF_Materiais_EstoqueMinimo DEFAULT 0,

    CONSTRAINT PK_Materiais
        PRIMARY KEY (Id),

    CONSTRAINT UQ_Materiais_Codigo
        UNIQUE (Codigo)
);
GO


/* ============================================================
   2. FUNCIONÁRIOS
   ============================================================ */

CREATE TABLE Funcionarios
(
    Id INT IDENTITY(1,1) NOT NULL,

    Nome NVARCHAR(150) NOT NULL,
    Matricula NVARCHAR(50) NOT NULL,
    CPF NVARCHAR(14) NOT NULL,
    Cargo NVARCHAR(100) NOT NULL,
    Idade INT NOT NULL,

    Telefone NVARCHAR(20) NULL,
    Setor NVARCHAR(100) NOT NULL,

    DataAdmissao DATETIME2 NOT NULL,

    Email NVARCHAR(150) NULL,
    Observacoes NVARCHAR(500) NULL,

    CONSTRAINT PK_Funcionarios
        PRIMARY KEY (Id),

    CONSTRAINT UQ_Funcionarios_Matricula
        UNIQUE (Matricula),

    CONSTRAINT UQ_Funcionarios_CPF
        UNIQUE (CPF)
);
GO


/* ============================================================
   3. MÁQUINAS
   ============================================================ */

CREATE TABLE Maquinas
(
    Id INT IDENTITY(1,1) NOT NULL,

    Nome NVARCHAR(150) NOT NULL,
    Codigo NVARCHAR(50) NOT NULL,
    Fabricante NVARCHAR(100) NOT NULL,
    Modelo NVARCHAR(100) NOT NULL,
    NumeroSerie NVARCHAR(100) NOT NULL,

    DataAquisicao DATETIME2 NOT NULL,

    CONSTRAINT PK_Maquinas
        PRIMARY KEY (Id),

    CONSTRAINT UQ_Maquinas_Codigo
        UNIQUE (Codigo),

    CONSTRAINT UQ_Maquinas_NumeroSerie
        UNIQUE (NumeroSerie)
);
GO


/* ============================================================
   4. TEMAS
   ============================================================ */

CREATE TABLE Temas
(
    Id INT IDENTITY(1,1) NOT NULL,

    Nome NVARCHAR(100) NOT NULL,
    Codigo NVARCHAR(50) NOT NULL,
    Ativo BIT NOT NULL
        CONSTRAINT DF_Temas_Ativo DEFAULT 1,

    CONSTRAINT PK_Temas
        PRIMARY KEY (Id),

    CONSTRAINT UQ_Temas_Codigo
        UNIQUE (Codigo)
);
GO


/* ============================================================
   5. CONFIGURAÇÃO DO SISTEMA
   ============================================================ */

CREATE TABLE ConfiguracaoSistema
(
    Id INT NOT NULL,
    TemaId INT NOT NULL,

    CONSTRAINT PK_ConfiguracaoSistema
        PRIMARY KEY (Id),

    CONSTRAINT FK_ConfiguracaoSistema_Temas
        FOREIGN KEY (TemaId)
        REFERENCES Temas(Id)
        ON DELETE NO ACTION
);
GO


/* ============================================================
   6. LOTES
   ============================================================ */

CREATE TABLE Lotes
(
    Id INT IDENTITY(1,1) NOT NULL,

    Codigo NVARCHAR(50) NOT NULL,

    Quantidade DECIMAL(18,2) NOT NULL,

    CustoUnitario DECIMAL(18,2) NOT NULL,

    DataEntrada DATETIME2 NOT NULL,

    DataFabricacao DATETIME2 NULL,
    DataValidade DATETIME2 NULL,

    MaterialId INT NOT NULL,

    Observacoes NVARCHAR(500) NULL,

    CONSTRAINT PK_Lotes
        PRIMARY KEY (Id),

    CONSTRAINT FK_Lotes_Materiais
        FOREIGN KEY (MaterialId)
        REFERENCES Materiais(Id)
        ON DELETE NO ACTION,

    CONSTRAINT UQ_Lotes_Material_Codigo
        UNIQUE (MaterialId, Codigo)
);
GO


/* ============================================================
   7. MOVIMENTAÇÕES
   ============================================================ */

CREATE TABLE Movimentacoes
(
    Id INT IDENTITY(1,1) NOT NULL,

    Tipo NVARCHAR(20) NOT NULL,

    Quantidade DECIMAL(18,2) NOT NULL,

    DataMovimentacao DATETIME2 NOT NULL,

    Observacoes NVARCHAR(500) NULL,

    MaterialId INT NOT NULL,

    LoteId INT NULL,

    CONSTRAINT PK_Movimentacoes
        PRIMARY KEY (Id),

    CONSTRAINT FK_Movimentacoes_Materiais
        FOREIGN KEY (MaterialId)
        REFERENCES Materiais(Id)
        ON DELETE NO ACTION,

    CONSTRAINT FK_Movimentacoes_Lotes
        FOREIGN KEY (LoteId)
        REFERENCES Lotes(Id)
        ON DELETE NO ACTION
);
GO


/* ============================================================
   8. ORDENS DE SERVIÇO
   ============================================================ */

CREATE TABLE OrdensServico
(
    Id INT IDENTITY(1,1) NOT NULL,

    NumeroOS NVARCHAR(50) NOT NULL,

    Cliente NVARCHAR(150) NOT NULL,

    Contato NVARCHAR(100) NULL,

    Endereco NVARCHAR(300) NULL,

    DataAbertura DATETIME2 NOT NULL,

    DescricaoServico NVARCHAR(1000) NOT NULL,

    TipoServico NVARCHAR(50) NOT NULL,

    DataEntrega DATETIME2 NULL,

    FuncionarioId INT NULL,

    MaquinaId INT NULL,

    ValorMateriais DECIMAL(18,2) NOT NULL
        CONSTRAINT DF_OS_ValorMateriais DEFAULT 0,

    ValorMaoObra DECIMAL(18,2) NOT NULL
        CONSTRAINT DF_OS_ValorMaoObra DEFAULT 0,

    Desconto DECIMAL(18,2) NOT NULL
        CONSTRAINT DF_OS_Desconto DEFAULT 0,

    ValorTotal DECIMAL(18,2) NOT NULL
        CONSTRAINT DF_OS_ValorTotal DEFAULT 0,

    CondicaoPagamento NVARCHAR(MAX) NULL,

    Observacoes NVARCHAR(500) NULL,

    CONSTRAINT PK_OrdensServico
        PRIMARY KEY (Id),

    CONSTRAINT UQ_OrdensServico_NumeroOS
        UNIQUE (NumeroOS),

    CONSTRAINT FK_OrdensServico_Funcionarios
        FOREIGN KEY (FuncionarioId)
        REFERENCES Funcionarios(Id)
        ON DELETE SET NULL,

    CONSTRAINT FK_OrdensServico_Maquinas
        FOREIGN KEY (MaquinaId)
        REFERENCES Maquinas(Id)
        ON DELETE SET NULL
);
GO


/* ============================================================
   9. ORDEM DE SERVIÇO + MATERIAIS
   ============================================================ */

CREATE TABLE OrdensServicoMateriais
(
    Id INT IDENTITY(1,1) NOT NULL,

    OrdemServicoId INT NOT NULL,

    MaterialId INT NOT NULL,

    Quantidade DECIMAL(18,2) NOT NULL,

    Unidade NVARCHAR(20) NOT NULL,

    ValorUnitario DECIMAL(18,2) NOT NULL,

    Subtotal DECIMAL(18,2) NOT NULL,

    CONSTRAINT PK_OrdensServicoMateriais
        PRIMARY KEY (Id),

    CONSTRAINT FK_OSMateriais_OrdensServico
        FOREIGN KEY (OrdemServicoId)
        REFERENCES OrdensServico(Id)
        ON DELETE CASCADE,

    CONSTRAINT FK_OSMateriais_Materiais
        FOREIGN KEY (MaterialId)
        REFERENCES Materiais(Id)
        ON DELETE NO ACTION
);
GO


/* ============================================================
   10. PROCESSOS DE PRODUÇÃO
   ============================================================ */

CREATE TABLE ProcessosProducao
(
    Id INT IDENTITY(1,1) NOT NULL,

    OrdemServicoId INT NOT NULL,

    MaquinaId INT NOT NULL,

    DataInicio DATETIME2 NOT NULL,

    DataFim DATETIME2 NULL,

    QuantidadePlanejada INT NOT NULL,

    QuantidadeProduzida INT NOT NULL,

    ProducaoPorMinuto DECIMAL(18,2) NOT NULL,

    ConsumoPorUnidade DECIMAL(18,4) NOT NULL,

    MaterialConsumido DECIMAL(18,4) NOT NULL,

    Status NVARCHAR(30) NOT NULL,

    CONSTRAINT PK_ProcessosProducao
        PRIMARY KEY (Id),

    CONSTRAINT FK_ProcessosProducao_OrdensServico
        FOREIGN KEY (OrdemServicoId)
        REFERENCES OrdensServico(Id)
        ON DELETE NO ACTION,

    CONSTRAINT FK_ProcessosProducao_Maquinas
        FOREIGN KEY (MaquinaId)
        REFERENCES Maquinas(Id)
        ON DELETE NO ACTION
);
GO


/* ============================================================
   11. CONFIGURAÇÕES DAS MÁQUINAS
   ============================================================ */

CREATE TABLE ConfiguracoesMaquina
(
    Id INT IDENTITY(1,1) NOT NULL,

    MaquinaId INT NOT NULL,

    TipoServico NVARCHAR(50) NOT NULL,

    ProducaoPorMinuto DECIMAL(18,2) NOT NULL,

    ConsumoPorUnidade DECIMAL(18,4) NOT NULL,

    Ativa BIT NOT NULL
        CONSTRAINT DF_ConfiguracoesMaquina_Ativa DEFAULT 1,

    CONSTRAINT PK_ConfiguracoesMaquina
        PRIMARY KEY (Id),

    CONSTRAINT FK_ConfiguracoesMaquina_Maquinas
        FOREIGN KEY (MaquinaId)
        REFERENCES Maquinas(Id)
        ON DELETE CASCADE
);
GO
