   10. RESUMO DOS DADOS
   ============================================================ */

FROM Funcionarios

SELECT 'Maquinas', COUNT(*)
FROM Maquinas

UNION ALL

SELECT 'Materiais', COUNT(*)
FROM Materiais

UNION ALL

SELECT 'Lotes', COUNT(*)
FROM Lotes

UNION ALL

SELECT 'Movimentacoes', COUNT(*)
FROM Movimentacoes

UNION ALL

SELECT 'OrdensServico', COUNT(*)
FROM OrdensServico

UNION ALL

SELECT 'OrdensServicoMateriais', COUNT(*)
FROM OrdensServicoMateriais

UNION ALL

FROM ProcessosProducao

UNION ALL

FROM ConfiguracoesMaquina;
